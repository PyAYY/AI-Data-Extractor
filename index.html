import os import sqlite3 import secrets import hashlib from typing import Optional from fastapi import FastAPI, HTTPException, Query, Header, Depends from fastapi.middleware.cors import CORSMiddleware from pydantic import BaseModel from google import genai from instructor import from_genai from instructor.processing.multimodal import Image from pydantic import Field app = FastAPI() app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"], ) DB_FILE = "saas_database.db" ADMIN_KEY = os.getenv("ADMIN_KEY", "master_secret_123") # Initialize Gemini & Instructor Client ai_client = genai.Client(api_key=os.getenv("GEMINI_API_KEY")) instructor_client = from_genai(ai_client) def init_db(): with sqlite3.connect(DB_FILE) as conn: c = conn.cursor() c.execute(""" CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, email TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL, api_key TEXT UNIQUE NOT NULL, access_token TEXT UNIQUE, credits INTEGER DEFAULT 10, tier TEXT DEFAULT 'free' ) """) conn.commit() init_db() def hash_password(password: str) -> str: return hashlib.sha256(password.encode()).hexdigest() # --- Pydantic Models --- class AuthRequest(BaseModel): email: str password: str class UpdateUserRequest(BaseModel): admin_secret: str target_api_key: str credits: int tier: str class ExtractionRequest(BaseModel): image_base64: str # --- Response Models for Extraction --- class InvoiceSchema(BaseModel): vendor_name: str = Field(description="Name of the company issuing the invoice") invoice_number: Optional[str] = Field(description="Invoice tracking number") total_amount: float = Field(description="Total monetary amount due") due_date: Optional[str] = Field(description="Payment due date") class ReceiptSchema(BaseModel): merchant_name: str = Field(description="Store or merchant name") transaction_date: Optional[str] = Field(description="Date of purchase") total_paid: float = Field(description="Total amount paid") items_purchased: list[str] = Field(description="List of items bought") class BusinessCardSchema(BaseModel): full_name: str = Field(description="Individual's full name") job_title: Optional[str] = Field(description="Professional title or role") company_name: Optional[str] = Field(description="Company or organization name") email: Optional[str] = Field(description="Contact email address") phone_number: Optional[str] = Field(description="Contact phone number") TASK_SCHEMAS = { "invoice": (InvoiceSchema, "Extract structured details from this invoice image."), "receipt": (ReceiptSchema, "Extract structured details from this store receipt image."), "business-card": (BusinessCardSchema, "Extract contact information from this business card image.") } #
